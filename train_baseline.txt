"""
Production-grade training pipeline for pure data-driven baseline model.

This script trains a BiLSTM model without physics constraints to serve as
the baseline comparison for the Robust PINN. The baseline demonstrates the
performance ceiling of pure data-driven approaches under sensor degradation.

Key Features:
    1. Compatible with refactored config.py (Dataclass hierarchy)
    2. Compatible with refactored dataset.py (5-element unpacking)
    3. Early stopping with patience mechanism
    4. Comprehensive logging and visualization
    5. Type-safe implementation with error handling

Mathematical Formulation:
    Baseline predicts position offsets: Δp̂ = f_θ(x_seq)
    Absolute position: p̂ = p_last + Δp̂
    Loss: L = MSE(p̂, p_target)
"""

# =============================================================================
# Baseline Model Architecture
# =============================================================================

class BaselineLSTM(nn.Module):
    """Pure data-driven BiLSTM baseline without physics constraints.

    This model serves as the comparison baseline to demonstrate the
    effectiveness of physics-informed regularization in the Robust PINN.

    Architecture:
        Input → BiLSTM → MLP Decoder → Position Offsets

    The model predicts position offsets relative to the last known position,
    which are then added to obtain absolute predictions.

    Attributes:
        pred_len: Prediction horizon length.
        n_targets: Number of output dimensions (3 for x, y, z).
        bilstm: Bidirectional LSTM encoder.
        decoder: MLP decoder for offset prediction.
    """

    def __init__(
        self,
        n_features: int,
        hidden_size: int = 128,
        num_layers: int = 2,
        pred_len: int = 5,
        dropout: float = 0.2
    ) -> None:
        super().__init__()
        self.pred_len = pred_len
        self.n_targets = 3

        self.bilstm = nn.LSTM(
            input_size=n_features,
            hidden_size=hidden_size,
            num_layers=num_layers,
            batch_first=True,
            bidirectional=True,
            dropout=dropout if num_layers > 1 else 0.0
        )

        lstm_out_dim = hidden_size * 2

        self.decoder = nn.Sequential(
            nn.Linear(lstm_out_dim, 128),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(128, 64),
            nn.ReLU(),
            nn.Linear(64, pred_len * self.n_targets)
        )

        self._init_weights()

    def _init_weights(self) -> None:
        """Initialize weights with Xavier uniform for stability."""
        for name, param in self.named_parameters():
            if 'weight' in name and param.dim() >= 2:
                nn.init.xavier_uniform_(param)
            elif 'bias' in name:
                nn.init.zeros_(param)

    def forward(self, x_seq: Tensor) -> Tensor:
        """Forward pass.

        Args:
            x_seq: Input sequence [B, T, F].

        Returns:
            Position offsets [B, pred_len, 3].
        """
        out, _ = self.bilstm(x_seq)
        last_hidden = out[:, -1, :]
        flat_output = self.decoder(last_hidden)
        offsets = flat_output.view(-1, self.pred_len, self.n_targets)
        return offsets


# =============================================================================
# Training History
# =============================================================================

@dataclass
class TrainingHistory:
    """Container for training metrics.

    Attributes:
        train_loss: Training loss per epoch.
        val_loss: Validation loss per epoch.
        learning_rates: Learning rate per epoch.
    """
    train_loss: List[float] = field(default_factory=list)
    val_loss: List[float] = field(default_factory=list)
    learning_rates: List[float] = field(default_factory=list)


# =============================================================================
# Training Functions
# =============================================================================

def train_one_epoch(
    model: BaselineLSTM,
    loader: DataLoader,
    optimizer: optim.Optimizer,
    device: torch.device,
    grad_clip_norm: float = 5.0
) -> float:
    """Execute one training epoch.

    Args:
        model: Baseline LSTM model.
        loader: Training data loader.
        optimizer: Optimizer.
        device: Computation device.
        grad_clip_norm: Maximum gradient norm for clipping.

    Returns:
        Average training loss for the epoch.
    """
    model.train()

    total_loss = 0.0
    n_batches = 0

    for batch in loader:
        x_seq, validity, target_pos, last_vel, last_pos = batch

        x_seq = x_seq.to(device, non_blocking=True)
        target_pos = target_pos.to(device, non_blocking=True)
        last_pos = last_pos.to(device, non_blocking=True)

        optimizer.zero_grad(set_to_none=True)

        pred_offset = model(x_seq)

        pred_pos = last_pos.unsqueeze(1) + pred_offset

        valid_mask = ~torch.isnan(target_pos)

        if valid_mask.any():
            loss = nn.functional.mse_loss(
                pred_pos[valid_mask],
                target_pos[valid_mask]
            )
        else:
            loss = torch.zeros(1, device=device, requires_grad=True)

        loss.backward()

        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=grad_clip_norm)

        optimizer.step()

        total_loss += loss.item()
        n_batches += 1

    return total_loss / max(n_batches, 1)


@torch.no_grad()
def validate(
    model: BaselineLSTM,
    loader: DataLoader,
    device: torch.device
) -> float:
    """Validate model on held-out data.

    Args:
        model: Baseline LSTM model.
        loader: Validation data loader.
        device: Computation device.

    Returns:
        Average validation loss.
    """
    model.eval()

    total_loss = 0.0
    n_batches = 0

    for batch in loader:
        x_seq, validity, target_pos, last_vel, last_pos = batch

        x_seq = x_seq.to(device, non_blocking=True)
        target_pos = target_pos.to(device, non_blocking=True)
        last_pos = last_pos.to(device, non_blocking=True)

        pred_offset = model(x_seq)

        pred_pos = last_pos.unsqueeze(1) + pred_offset

        valid_mask = ~torch.isnan(target_pos)

        if valid_mask.any():
            loss = nn.functional.mse_loss(
                pred_pos[valid_mask],
                target_pos[valid_mask]
            )
        else:
            loss = torch.zeros(1, device=device)

        total_loss += loss.item()
        n_batches += 1

    return total_loss / max(n_batches, 1)


# =============================================================================
# Visualization
# =============================================================================

def plot_learning_curve(history: TrainingHistory, save_path: Path) -> None:
    """Generate learning curve visualization.

    Args:
        history: Training history container.
        save_path: Path to save figure.
    """
    epochs = range(1, len(history.train_loss) + 1)

    fig, ax = plt.subplots(figsize=(10, 6))

    ax.plot(epochs, history.train_loss, label='Training Loss',
            color='#3498DB', linewidth=2, marker='o', markersize=3,
            markevery=max(1, len(epochs) // 20))
    ax.plot(epochs, history.val_loss, label='Validation Loss',
            color='#E74C3C', linewidth=2, linestyle='--', marker='s',
            markersize=3, markevery=max(1, len(epochs) // 20))

    ax.set_xlabel('Epoch', fontsize=12)
    ax.set_ylabel('MSE Loss', fontsize=12)
    ax.set_title('Baseline LSTM Learning Curve', fontsize=14, fontweight='bold')
    ax.legend(loc='upper right', fontsize=11, framealpha=0.95)
    ax.grid(True, alpha=0.3, linewidth=0.5)
    ax.set_yscale('log')

    best_epoch = np.argmin(history.val_loss) + 1
    best_val_loss = min(history.val_loss)
    ax.axvline(x=best_epoch, color='gray', linestyle=':', linewidth=1.5,
               label=f'Best Epoch: {best_epoch}')
    ax.scatter([best_epoch], [best_val_loss], color='red', s=100,
               zorder=5, marker='*', edgecolors='black', linewidths=1.5)

    ax.legend(loc='upper right', fontsize=11, framealpha=0.95)

    plt.tight_layout()
    save_path.parent.mkdir(parents=True, exist_ok=True)
    plt.savefig(save_path, dpi=200, bbox_inches='tight', facecolor='white')
    plt.close()

    logger.info(f"Learning curve saved: {save_path}")


# =============================================================================
# Checkpoint Management
# =============================================================================

def save_checkpoint(
    path: Path,
    epoch: int,
    model: BaselineLSTM,
    optimizer: optim.Optimizer,
    scheduler: optim.lr_scheduler._LRScheduler,
    val_loss: float,
    history: TrainingHistory
) -> None:
    """Save training checkpoint.

    Args:
        path: Checkpoint file path.
        epoch: Current epoch number.
        model: Model to save.
        optimizer: Optimizer state.
        scheduler: LR scheduler state.
        val_loss: Current validation loss.
        history: Training history.
    """
    checkpoint = {
        'epoch': epoch,
        'model_state': model.state_dict(),
        'optimizer_state': optimizer.state_dict(),
        'scheduler_state': scheduler.state_dict() if scheduler else None,
        'val_loss': val_loss,
        'train_loss_history': history.train_loss,
        'val_loss_history': history.val_loss,
        'lr_history': history.learning_rates,
    }

    path.parent.mkdir(parents=True, exist_ok=True)
    torch.save(checkpoint, path)


# =============================================================================
# Main Training Pipeline
# =============================================================================

def main() -> None:
    """Main training entry point."""
    cfg = Config()
    cfg.setup_environment(verbose=True)

    device = torch.device(cfg.DEVICE)

    logger.info("=" * 70)
    logger.info("Baseline LSTM Training Pipeline")
    logger.info("=" * 70)
    logger.info(f"Device: {device}")

    dataset_config = DatasetConfig(
        seq_len=cfg.SEQ_LEN,
        pred_len=cfg.PRED_LEN,
        train_ratio=cfg.TRAIN_RATIO,
        val_ratio=getattr(cfg, 'VAL_RATIO', 0.15),
        batch_size=cfg.BATCH_SIZE,
        feature_cols=tuple(cfg.FEATURE_COLS),
        target_cols=tuple(cfg.TARGET_COLS),
        vel_cols=tuple(cfg.VEL_COLS)
    )

    logger.info("Loading datasets...")
    pipeline = AUVDataPipeline(cfg.GT_PATH, cfg.COR_PATH, dataset_config)
    train_loader, val_loader, _ = pipeline.get_dataloaders(num_workers=0, pin_memory=False)

    model = BaselineLSTM(
        n_features=cfg.N_FEATURES,
        hidden_size=getattr(cfg, 'HIDDEN_SIZE', 128),
        num_layers=getattr(cfg, 'NUM_LAYERS', 2),
        pred_len=cfg.PRED_LEN,
        dropout=cfg.DROPOUT
    ).to(device)

    n_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
    logger.info(f"Model parameters: {n_params:,}")

    optimizer = optim.AdamW(
        model.parameters(),
        lr=cfg.LR,
        weight_decay=getattr(cfg, 'WEIGHT_DECAY', 1e-4)
    )

    scheduler = optim.lr_scheduler.CosineAnnealingWarmRestarts(
        optimizer,
        T_0=getattr(cfg, 'LR_T0', 50),
        T_mult=getattr(cfg, 'LR_TMULT', 2),
        eta_min=cfg.LR * 0.01
    )

    history = TrainingHistory()
    best_val_loss = float('inf')
    patience_counter = 0
    patience = 15

    best_ckpt_path = cfg.SAVE_DIR / 'baseline_best.pth'
    last_ckpt_path = cfg.SAVE_DIR / 'baseline_last.pth'

    logger.info("")
    logger.info("=" * 70)
    logger.info(f"{'Epoch':>6} │ {'Train Loss':>12} │ {'Val Loss':>12} │ {'LR':>10} │ {'Time':>6}")
    logger.info("=" * 70)

    total_epochs = cfg.EPOCHS
    start_time = time.time()

    try:
        for epoch in range(1, total_epochs + 1):
            epoch_start = time.time()

            train_loss = train_one_epoch(
                model=model,
                loader=train_loader,
                optimizer=optimizer,
                device=device,
                grad_clip_norm=getattr(cfg, 'GRAD_CLIP', 5.0)
            )

            val_loss = validate(
                model=model,
                loader=val_loader,
                device=device
            )

            scheduler.step()

            history.train_loss.append(train_loss)
            history.val_loss.append(val_loss)
            history.learning_rates.append(optimizer.param_groups[0]['lr'])

            epoch_time = time.time() - epoch_start

            if epoch == 1 or epoch % 10 == 0 or epoch == total_epochs:
                lr_now = optimizer.param_groups[0]['lr']
                logger.info(
                    f"{epoch:>6} │ {train_loss:>12.6f} │ {val_loss:>12.6f} │ "
                    f"{lr_now:>10.2e} │ {epoch_time:>5.1f}s"
                )

            if val_loss < best_val_loss:
                best_val_loss = val_loss
                patience_counter = 0

                save_checkpoint(
                    path=best_ckpt_path,
                    epoch=epoch,
                    model=model,
                    optimizer=optimizer,
                    scheduler=scheduler,
                    val_loss=val_loss,
                    history=history
                )
            else:
                patience_counter += 1

            if patience_counter >= patience:
                logger.info("")
                logger.info(f"Early stopping triggered after {patience} epochs without improvement")
                break

    except KeyboardInterrupt:
        logger.info("")
        logger.info("Training interrupted by user")

    except Exception as e:
        logger.error(f"Training failed with error: {e}")
        raise

    finally:
        save_checkpoint(
            path=last_ckpt_path,
            epoch=epoch,
            model=model,
            optimizer=optimizer,
            scheduler=scheduler,
            val_loss=val_loss,
            history=history
        )

    total_time = time.time() - start_time
    logger.info("=" * 70)
    logger.info("")
    logger.info("Training Complete!")
    logger.info(f"  Total time: {total_time/60:.1f} minutes")
    logger.info(f"  Best validation loss: {best_val_loss:.6f}")
    logger.info(f"  Best model saved to: {best_ckpt_path}")

    logger.info("")
    logger.info("Generating learning curve...")
    plot_learning_curve(history, cfg.FIG_DIR / 'baseline_learning_curve.png')

    logger.info("")
    logger.info("All done!")


if __name__ == '__main__':
    main()
